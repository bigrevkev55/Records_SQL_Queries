--Purpose: The query seeks to provide all active rooms on the Clarksville Campus

SELECT slbrdef_bldg_code,
       slbrdef_room_number,
       slbrdef_desc,
       slbrdef_maximum_capacity,
       slbrdef_room_type,
       slbrdef_term_code_eff,
       slbrdef_user_id,
       slbrdef_rmst_code as "Room Status",
       rn
FROM (
  SELECT slbrdef_bldg_code,
         slbrdef_room_number,
         slbrdef_desc,
         slbrdef_maximum_capacity,
         slbrdef_room_type,
         slbrdef_term_code_eff,
         slbrdef_user_id,
         slbrdef_rmst_code,
         ROW_NUMBER() OVER (
           PARTITION BY slbrdef_bldg_code, slbrdef_room_number
           ORDER BY slbrdef_term_code_eff DESC
         ) AS rn
  FROM slbrdef
  WHERE slbrdef_bldg_code IN ('CLARK', 'CLRKA')
) 
WHERE rn = 1;
