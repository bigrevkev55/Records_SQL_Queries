select distinct ssrmeet_room_code,  SLBRDEF_DESC, SLBRDEF_CAPACITY, slbrdef.slbrdef_bldg_code
from ssrmeet
join slbrdef on slbrdef_room_number = ssrmeet_room_code
where slbrdef_rmst_code = 'AC'
order by ssrmeet_room_code;
