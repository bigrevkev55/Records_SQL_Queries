--Purpose: This script counts records received dates on SOAPCOL for statistical analysis of division of work. Update date range and last name starting letter (lines 5-6). Remove all lines below line 6 to get a count inclusive of TBR. Replace AND NOT with AND on line 7 plus insert ( before SORPCOL.SORPCOL_SGBI_CODE and replace AND NOT with OR from all lines below 8 to get TBR counts.


SELECT Count(SORPCOL.SORPCOL_PIDM)
FROM SPRIDEN,
SORPCOL
WHERE SORPCOL.SORPCOL_PIDM  = SPRIDEN.SPRIDEN_PIDM
AND SORPCOL.SORPCOL_TRANS_REV_DATE between '01-JAN-2022' and '30-DEC-2022'
--AND SPRIDEN.SPRIDEN_LAST_NAME LIKE 'A%'
AND NOT SORPCOL.SORPCOL_SBGI_CODE = '001028' 
--APSU
AND NOT SORPCOL.SORPCOL_SBGI_CODE = '001198' 
--ETSU
AND NOT SORPCOL.SORPCOL_SBGI_CODE = '001459' 
--UMEMPHIS
AND NOT SORPCOL.SORPCOL_SBGI_CODE = '001881' 
--VSCC
AND NOT SORPCOL.SORPCOL_SBGI_CODE = '000274' 
--SWTCC
AND NOT SORPCOL.SORPCOL_SBGI_CODE = '000319' 
--PSCC
AND NOT SORPCOL.SORPCOL_SBGI_CODE = '000453' 
--NESCC
AND NOT SORPCOL.SORPCOL_SBGI_CODE = '001081' 
--CSCC
AND NOT SORPCOL.SORPCOL_SBGI_CODE = '001084' 
--CHSCC
AND NOT SORPCOL.SORPCOL_SBGI_CODE = '001466' 
--MTSU
AND NOT SORPCOL.SORPCOL_SBGI_CODE = '001543' 
--MSCC
AND NOT SORPCOL.SORPCOL_SBGI_CODE = '001656' 
--RSCC
AND NOT SORPCOL.SORPCOL_SBGI_CODE = '001803' 
--TSU
AND NOT SORPCOL.SORPCOL_SBGI_CODE = '001804' 
--TTU
AND NOT SORPCOL.SORPCOL_SBGI_CODE = '001893' 
--WSCC
AND NOT SORPCOL.SORPCOL_SBGI_CODE = '002266' 
--JSCC
AND NOT SORPCOL.SORPCOL_SBGI_CODE = '002848' 
--CLSCC
