--Purpose: This script returns all high schools/colleges with code and address for import into Slate.

SELECT 
SOBSBGI.SOBSBGI_SBGI_CODE AS CODE,
STVSBGI_DESC AS NAME,
STVSBGI_TYPE_IND AS TYPE,
SOBSBGI_STREET_LINE1 AS STREET1,
SOBSBGI_STREET_LINE2 AS STREET2,
SOBSBGI_CITY AS CITY,
SOBSBGI_STAT_CODE  AS REGION,
SOBSBGI_ZIP AS POSTAL
FROM 
SOBSBGI
RIGHT JOIN STVSBGI ON SOBSBGI.SOBSBGI_SBGI_CODE = STVSBGI.STVSBGI_CODE
