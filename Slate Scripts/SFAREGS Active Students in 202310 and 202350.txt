--Purpose: This query returns all students with registration in 202310 and 202350 for import into 
--         Slate.  This will be used to email all students about submitting intents to graduate and 
--         cap and gown ordering


--desc goremal;

select spriden_first_name as "First Name",
       spriden_last_name as "Last Name",
       listAgg(GOREMAL_EMAIL_ADDRESS, ';') WITHIN GROUP (order by GOREMAL_EMAIL_ADDRESS)  as "Email Address"
    
from spriden
outer join goremal on spriden_pidm = goremal_pidm

where
spriden_change_ind is NULL
and goremal_status_ind = 'A'
and goremal_preferred_ind = 'Y'
and spriden_pidm in (
select distinct sfrstcr_pidm from sfrstcr where sfrstcr_term_code in ('202310', '202350')
)
group by spriden_first_name, spriden_last_name
order by spriden_last_name;
