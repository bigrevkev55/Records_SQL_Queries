-- Purpose: Returns VA Students and their programs of study (major & concentration) for a given VA cert term.
-- Edits:   11-MAR-2026 added in SSN and Major/Concentration Descriptions...kt

SELECT
    s.spriden_id                      AS student_id,
    p.spbpers_ssn                     AS ssn,
    s.spriden_last_name               AS last,
    s.spriden_first_name              AS first,
    v.sgrvetn_cert_hours              AS hours_certified,
    st.sgbstdn_degc_code_1            AS degree_type,
    mj.stvmajr_desc                   AS major,
    cn.stvmajr_desc                   AS concentration,
    v.sgrvetn_term_code_va            AS cert_term,
    vc.stvvetc_desc                   AS vet_code
FROM spriden s
JOIN sgrvetn v
  ON v.sgrvetn_pidm = s.spriden_pidm
JOIN spbpers p
  ON p.spbpers_pidm = s.spriden_pidm
JOIN sgbstdn st
  ON st.sgbstdn_pidm = s.spriden_pidm
JOIN stvvetc vc
  ON vc.stvvetc_code = v.sgrvetn_vetc_code
LEFT JOIN stvmajr mj
  ON mj.stvmajr_code = st.sgbstdn_majr_code_1
LEFT JOIN stvmajr cn
  ON cn.stvmajr_code = st.sgbstdn_majr_code_conc_1
WHERE
    v.sgrvetn_term_code_va = :TERM
    AND v.sgrvetn_vetc_code IS NOT NULL
    AND s.spriden_change_ind IS NULL
    AND st.sgbstdn_term_code_eff = (
        SELECT MAX(st2.sgbstdn_term_code_eff)
        FROM sgbstdn st2
        WHERE st2.sgbstdn_pidm = st.sgbstdn_pidm
          AND st2.sgbstdn_term_code_eff <= :TERM
    )
ORDER BY
    v.sgrvetn_cert_hours,
    s.spriden_last_name,
    s.spriden_first_name,
    v.sgrvetn_term_code_va;
