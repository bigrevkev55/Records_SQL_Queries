--   Purpose:
--   This query identifies students with a Veterans Deferment
--   (TZRSTSF_STATUS_CODE = 'V') in the selected term who are
--   enrolled in credit-bearing coursework but do not have a
--   Veterans Benefit Type (SGRVETN_VETC_CODE) assigned for the
--   same term.
--
--   Prompts:
--   1. When prompted for "Term", enter the term code to be audited.
--
--   Edits:
--   1. Original query revised September 2026 to identify students
--      with a TZRSTSF Veterans Deferment who are missing a VA
--      Benefit Type in SGRVETN for the audited term.
--   2. Query now returns the TZRSTSF term rather than SGRVETN term
--      information because students being identified may not have
--      an SGRVETN record for the selected term.
--   3. Testing determined that using the maximum TZRSTSF_ACTIVITY_DATE
--      excluded valid students with active Veterans Deferments.
--      Therefore, the query identifies students with a V status in
--      TZRSTSF for the selected term without requiring the most
--      recent TZRSTSF record.
--   4. Warning: A student returned by this query has a TZRSTSF
--      Veterans Deferment in the prompted term; however, the V
--      status may not appear as the first or most recent record
--      displayed in TZASTSF/TZRSTSF history. Review the complete
--      TZRSTSF record history before making updates.


--desc TZRSTSF;
--desc spriden;
--desc sgrvetn;

select distinct
       spriden_id as "ID",
       tzrstsf_term as "TZA Term",
       spriden_last_name as "Last Name",
       spriden_first_name as "First Name"
from spriden,
     tzrstsf
where spriden_pidm = tzrstsf_pidm
      and spriden_change_ind is null
      and tzrstsf_term = :Term
      and tzrstsf_status_code = 'V'
      and spriden_pidm in
            (select distinct sfrstcr_pidm
               from sfrstcr
              where sfrstcr_term_code = :Term
                and sfrstcr_credit_hr > 0)
      and not exists
            (select 1
               from sgrvetn
              where sgrvetn_pidm = spriden_pidm
                and sgrvetn_term_code_va = :Term
                and sgrvetn_vetc_code is not null)
order by spriden_last_name,
         spriden_first_name;
