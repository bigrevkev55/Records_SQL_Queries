/*                  ---This code returns the most recent prereq from SMAAREA

select SMRACAA_AREA, SMRACAA_TERM_CODE_EFF,SMRACAA_SET, SMRACAA_SUBSET, SMRACAA_RULE,SMRACAA_SUBJ_CODE,
       SMRACAA_CRSE_NUMB_LOW, SMRACAA_CRSE_NUMB_HIGH, SMRACAA_ATTR_CODE,SMRACAA_GRDE_CODE_MIN, SMRACAA_CONCURRENCY_IND,
       SMRACAA_TESC_CODE, SMRACAA_MIN_VALUE, SMRACAA_MAX_VALUE
from SMRACAA o 
where smracaa_area = :SMAALIB_CODE 
      and SMRAcaa_term_code_eff = (
          select max(SMRAcaa_term_code_eff) 
          from smracaa i 
          where o.smracaa_area=i.smracaa_area
        )
order by 3,4;
*/

select SSBSECT_term_code as "Term",
       SSBSECT_PTRM_CODE as "Part of Term",
       SSBSECT_CRN as "CRN",
       SSBSECT_SEQ_NUMB as "Section Code",
       SSBSECT_SUBJ_CODE as "Subject",
       SSBSECT_CRSE_NUMB as "Course",
       SSRRARE_AREA as "Prereq"
from ssbsect
left outer join ssrrare on ssrrare_term_code = ssbsect_term_code 
                       and ssrrare_crn = ssbsect_crn
where ssbsect_term_code = :Term_Code
and ssbsect_ptrm_code = :Part_of_Term_Code
and SSBSECT_SUBJ_CODE = :Subject_Code
order by SSBSECT_SUBJ_CODE, SSBSECT_CRSE_NUMB, SSBSECT_SEQ_NUMB;
