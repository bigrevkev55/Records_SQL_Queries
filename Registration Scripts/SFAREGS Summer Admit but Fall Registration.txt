/***** SORLCUR Version *****/
SELECT iden.SPRIDEN_ID AS STUDENT_ID,
    chrt.FA_ENR,
    chrt.SU_ENR,
    lcur.*
FROM (
-- Valid Term Enrollment
    SELECT SFRSTCR_PIDM,
        MAX(CASE WHEN SFRSTCR_TERM_CODE IN ('202480') THEN 'Y' ELSE 'N' END) AS FA_ENR,
        MAX(CASE WHEN SFRSTCR_TERM_CODE IN ('202450') THEN 'Y' ELSE 'N' END) AS SU_ENR
    FROM SFRSTCR
        LEFT JOIN STVRSTS ON SFRSTCR_RSTS_CODE = STVRSTS_CODE
    WHERE SFRSTCR_LEVL_CODE NOT IN ('VO', 'CE')
        AND SFRSTCR_CREDIT_HR > 0
        AND STVRSTS_INCL_SECT_ENRL = 'Y'
    GROUP BY SFRSTCR_PIDM
) chrt
-- Student Curriculum
LEFT JOIN (
    SELECT aa.SORLCUR_PIDM,
        MAX(SORLCUR_TERM_CODE_ADMIT) ADMIT_TERM
    FROM SORLCUR aa
    WHERE aa.SORLCUR_TERM_CODE = (
        SELECT MAX(bb.SORLCUR_TERM_CODE)
        FROM SORLCUR bb
        WHERE aa.SORLCUR_PIDM = bb.SORLCUR_PIDM
    )
    GROUP BY aa.SORLCUR_PIDM
) lcur ON chrt.SFRSTCR_PIDM = lcur.SORLCUR_PIDM
-- Student Identifiers
LEFT JOIN SPRIDEN iden ON chrt.SFRSTCR_PIDM = iden.SPRIDEN_PIDM
    AND iden.SPRIDEN_CHANGE_IND IS NULL
-- Report Organization
WHERE chrt.FA_ENR = 'Y'
    AND chrt.SU_ENR = 'N'
    AND lcur.ADMIT_TERM = '202450'
;

/***** SARADAP Version *****/
SELECT iden.SPRIDEN_ID AS STUDENT_ID,
    chrt.SUMMER_ENROLLMENT,
    chrt.FALL_ENROLLMENT,
    adap.SUMMER_APPLICATION,
    adap.FALL_APPLICATION,
    stdn.SGBSTDN_TERM_CODE_ADMIT
FROM (
-- Valid Enrollment
    SELECT SFRSTCR_PIDM,
        MAX(CASE WHEN SFRSTCR_TERM_CODE = '202450' THEN 'Y' ELSE 'N' END) AS SUMMER_ENROLLMENT,
        MAX(CASE WHEN SFRSTCR_TERM_CODE = '202480' THEN 'Y' ELSE 'N' END) AS FALL_ENROLLMENT
    FROM SFRSTCR
        LEFT JOIN STVRSTS ON SFRSTCR_RSTS_CODE = STVRSTS_CODE
    WHERE SFRSTCR_LEVL_CODE NOT IN ('VO', 'CE')
        AND SFRSTCR_CREDIT_HR > 0
        AND STVRSTS_INCL_SECT_ENRL = 'Y'
    GROUP BY SFRSTCR_PIDM
) chrt
LEFT JOIN (
-- Student Applications
    SELECT SARADAP_PIDM,
        MAX(CASE WHEN SARADAP_TERM_CODE_ENTRY = '202450' THEN 'Y' ELSE 'N' END) AS SUMMER_APPLICATION,
        MAX(CASE WHEN SARADAP_TERM_CODE_ENTRY = '202480' THEN 'Y' ELSE 'N' END) AS FALL_APPLICATION
    FROM SARADAP
    GROUP BY SARADAP_PIDM
) adap ON chrt.SFRSTCR_PIDM = adap.SARADAP_PIDM
-- Student Major
LEFT JOIN SGBSTDN stdn ON chrt.SFRSTCR_PIDM = stdn.SGBSTDN_PIDM
    AND stdn.SGBSTDN_TERM_CODE_EFF = (
        SELECT MAX(bb.SGBSTDN_TERM_CODE_EFF)
        FROM SGBSTDN bb
        WHERE chrt.SFRSTCR_PIDM = bb.SGBSTDN_PIDM
    )
-- Student Identifiers
LEFT JOIN SPRIDEN iden ON chrt.SFRSTCR_PIDM = iden.SPRIDEN_PIDM
    AND iden.SPRIDEN_CHANGE_IND IS NULL
-- Report Organization
WHERE SUMMER_ENROLLMENT = 'N'
    AND FALL_ENROLLMENT = 'Y'
    AND SUMMER_APPLICATION = 'Y'
    AND FALL_APPLICATION = 'N'
    AND stdn.SGBSTDN_TERM_CODE_ADMIT NOT LIKE ('%80')
;
