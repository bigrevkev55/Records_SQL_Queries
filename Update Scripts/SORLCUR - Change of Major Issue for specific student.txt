--Purpose: This script will insert end terms for a student who has 29 curriculum 
--         records with no end term.  To do this manually would be much more labor 
--         intensive.  


UPDATE sorlcur
SET sorlcur_term_code_end = '200580'
WHERE sorlcur_pidm = '123188'
               AND SORLCUR_SEQNO = '44';

--Purpose: This script will insert end terms for a specific student who has one program without an end term or 
--         inactive status.

WHERE sorlcur_pidm = '76024' 
      AND sorlcur_seqno = '46';
