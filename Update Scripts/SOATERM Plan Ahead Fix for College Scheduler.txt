-- UPDATE SQL for SQL Data Change Form to update the Background table via SQL
UPDATE sobterm
SET sobterm_max_plans = 1,
    sobterm_plan_term_open_cde = 'Y',
    sobterm_plan_reg_cde = 'Y'
WHERE sobterm_term_code = '202610';


-- Script to check setup
Select sobterm_max_plans, sobterm_plan_term_open_cde, sobterm_plan_reg_cde
from sobterm
where sobterm_term_code = :Registration_Term;
