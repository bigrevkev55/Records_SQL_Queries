/*Script #3 - Add the attribute to the proper sections for 202480*/
declare
    v_crn       varchar2(5);
    v_count     number := 0;
cursor  c_get_crn is
    select  distinct ssrattr_crn
    from    ssrattr
    --join to limit selection by ssbsect data
    join    ssbsect on  ssrattr_crn = ssbsect_crn
                    and ssrattr_term_code = '202480'
                    and ssbsect_term_code = ssrattr_term_code
    --join to limit selection by scrattr data
    join    scrattr on  ssbsect_subj_code || ssbsect_crse_numb = scrattr_subj_code || scrattr_crse_numb 
                    and scrattr_eff_term = '202480'
                    and scrattr_attr_code = 'WOR' 
;
begin
	open c_get_crn;
        loop
        fetch c_get_crn into v_crn;
        exit when c_get_crn%notfound;
            insert into ssrattr (
                                    ssrattr_term_code
                                ,   ssrattr_crn
                                ,   ssrattr_attr_code
                                ,   ssrattr_activity_date
                                ,   ssrattr_version
                                ,   ssrattr_user_id
                                )
            values  (
                        '202480'
                    ,   v_crn
                    ,   'WOR'
                    ,   sysdate
                    ,   '0'
                    ,   'BANWORX'
                )
            ;       
            v_count := v_count + 1;
        end loop; 
    close c_get_crn;
    commit;	
end;
