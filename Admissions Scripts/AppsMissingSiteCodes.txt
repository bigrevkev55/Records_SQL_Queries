--Purpose: This script returns all students who have missing site code on SAAADMS along with the answers from SAAQUAN.


SELECT SPRIDEN.SPRIDEN_ID        AS SCHOOL_ID,
  SPRIDEN.SPRIDEN_LAST_NAME      AS LAST_NAME,
  SPRIDEN.SPRIDEN_FIRST_NAME     AS FIRST_NAME,
  SARQUAN.SARQUAN_QUESTION      AS APP_QUESTIONS,
  SARQUAN.SARQUAN_ANSWER        AS APP_ANSWERS
FROM SPRIDEN,
SARQUAN,
SARADAP
WHERE SARQUAN.SARQUAN_PIDM  = SPRIDEN.SPRIDEN_PIDM
AND SARADAP.SARADAP_PIDM  = SPRIDEN.SPRIDEN_PIDM
and spriden_change_ind is NULL
AND SARADAP.SARADAP_TERM_CODE_ENTRY = '202310'
AND SARADAP.SARADAP_SITE_CODE IS NULL;
