--Purpose: This script returns all students who have NSCC (000850) listed on SOAPCOL.

SELECT SPRIDEN.SPRIDEN_ID        AS SCHOOL_ID,
  SPRIDEN.SPRIDEN_LAST_NAME      AS LAST_NAME,
  SPRIDEN.SPRIDEN_FIRST_NAME     AS FIRST_NAME,
  SORPCOL.SORPCOL_SBGI_CODE
FROM SPRIDEN,
SORPCOL
WHERE SORPCOL.SORPCOL_PIDM  = SPRIDEN.SPRIDEN_PIDM
and spriden_change_ind is NULL
AND SORPCOL.SORPCOL_SBGI_CODE = '000850';
