--Purpose: This script returns all dual enrollment students under 18 who have a SAAADMS residency code of 1 but have a residency code of Y on SFAREGS.

SELECT SPRIDEN.SPRIDEN_ID        AS SCHOOL_ID,
  SGBSTDN_TERM_CODE_EFF,
  SGBSTDN_RESD_CODE,
  SARADAP.SARADAP_RESD_CODE,
  SGBSTDN_ADMT_CODE
FROM SPRIDEN,
SGBSTDN,
SARADAP,
SPBPERS
WHERE SGBSTDN.SGBSTDN_PIDM  = SPRIDEN.SPRIDEN_PIDM
AND SARADAP.SARADAP_PIDM  = SPRIDEN.SPRIDEN_PIDM
AND SPBPERS.SPBPERS_PIDM  = SPRIDEN.SPRIDEN_PIDM
AND spriden_change_ind is NULL
AND SARADAP.SARADAP_TERM_CODE_ENTRY = '202310'
--update term
AND SGBSTDN_TERM_CODE_EFF = '202310'
AND SGBSTDN_RESD_CODE = 'Y'
AND SGBSTDN_ADMT_CODE = 'DE'
--update to Z to check for those as well
AND SARADAP.SARADAP_RESD_CODE = '1'
--update date to find under 18s
AND SPBPERS.SPBPERS_BIRTH_DATE < '13-JAN-2005';
