SELECT
    SHBTEOT.SHBTEOT_LAST_NAME      AS LAST,
    SHBTEOT.SHBTEOT_FIRST_NAME     AS FIRST,
    SHBTEOT.SHBTEOT_BANNER_ID      AS A_NUMBER,
    SHRTEOD.SHRTEOD_EMAIL_ADDR     AS E_MAIL,
    S.SHRTEOS_ORDER_ID             AS ORDER_NUMBER,
    S.SHRTEOS_ETST_CODE            AS STATUS_CODE,
    S.SHRTEOS_ETST_DATE            AS STATUS_DATE,
    S.SHRTEOS_VERSION,
    SHBTEOT.SHBTEOT_ATTACHMENT_IND AS ATTACHMENT,
    SHBTEOT.SHBTEOT_CANCEL_IND     AS CANCELLED,
    SHRTEOD.SHRTEOD_RECIPIENT_NAME AS RECIPIENT
FROM SHRTEOS S

JOIN SHBTEOT 
  ON SHBTEOT.SHBTEOT_ORDER_ID = S.SHRTEOS_ORDER_ID

JOIN SHRTEOD 
  ON SHRTEOD.SHRTEOD_ORDER_ID = S.SHRTEOS_ORDER_ID

WHERE 
    -- ✅ TRUE latest row (not VERSION)
    S.SHRTEOS_SURROGATE_ID = (
        SELECT MAX(S2.SHRTEOS_SURROGATE_ID)
        FROM SHRTEOS S2
        WHERE S2.SHRTEOS_ORDER_ID = S.SHRTEOS_ORDER_ID
    )

    -- ✅ latest status must be RG
    AND S.SHRTEOS_ETST_CODE = 'RG'

    -- ✅ optional date filter
    AND S.SHRTEOS_ETST_DATE > DATE '2026-05-15'

ORDER BY 
    SHBTEOT.SHBTEOT_LAST_NAME,
    SHBTEOT.SHBTEOT_FIRST_NAME;
