select spriden_id as "A Number",
       spriden_last_name as "Last Name",
       spriden_first_name as "First Name",
       --SORHSCH_SBGI_CODE as "School Code",
       STVSBGI_DESC      as "School"--,
---                          as "Phone"
from spriden, stvsbgi, sgbstdn, sorhsch

where spriden_change_ind is NULL
      and sgbstdn_pidm = spriden_pidm
      and spriden_pidm = sorhsch_pidm
      and sgbstdn_term_code_eff = '202210'
      and SGBSTDN_STYP_CODE = 'H'
      and SGBSTDN_RATE_CODE = 'DUAL'
      and stvsbgi_code = sorhsch_sbgi_code
      and spriden_pidm in (select distinct sfrstcr_pidm from sfrstcr where sfrstcr_term_code = '202210')

order by STVSBGI_DESC, spriden_last_name, spriden_first_name;
