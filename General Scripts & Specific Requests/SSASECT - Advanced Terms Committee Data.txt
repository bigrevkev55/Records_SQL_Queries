SELECT
    ssbsect_crn       crn,
    ssbsect_subj_code subj,
    ssbsect_crse_numb course,
    ssbsect_enrl      enrollment,
    ssbsect_insm_code instructional_method,
    ssrmeet_days_code "Days"
FROM
    ssrmeet,
    ssbsect
WHERE
        ssrmeet_term_code = '202280'
    AND ssbsect_term_code = '202280'
    AND ssrmeet_crn = ssbsect_crn
    AND ssrmeet_sat_day IS NOT NULL
    AND ssrmeet_sat_day IS NOT NULL
    AND ssbsect_subj_code != 'NSCC';
