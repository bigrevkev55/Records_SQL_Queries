--         hours for 202210.  These will have to be manually fixed.  


--desc sfbetrm;

select spriden_id as "A Number",
       spriden_last_name as "Last",
       spriden_first_name as "First",
       sfbetrm_mhrs_over as "Max Hours"
from spriden
inner join sfbetrm on sfbetrm_pidm=spriden_pidm

Where sfbetrm_mhrs_over > '21'
      and SFBETRM_TERM_CODE = '202510'
      and spriden_change_ind IS NULL
      and exists(select sfrstcr_pidm from sfrstcr where sfrstcr_term_code = 202510)

Order by 2,1;
