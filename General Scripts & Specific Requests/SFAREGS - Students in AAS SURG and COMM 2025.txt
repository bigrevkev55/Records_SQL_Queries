--Purpose:  This query returns active students in the AAS_SURG program
--          who are enrolled in COMM 2025 in term 202580.

SELECT
    pers.spriden_id AS "A Number",
    pers.spriden_last_name AS "Last Name",
    pers.spriden_first_name AS "First Name",
    prgm.sgbstdn_term_code_eff AS "Term Updated",
    prgm.sgbstdn_term_code_ctlg_1 AS "Catalog Year",
    prgm.sgbstdn_program_1 AS "Program"
FROM
    SGBSTDN prgm
JOIN
    SPRIDEN pers ON prgm.sgbstdn_pidm = pers.spriden_pidm
JOIN
    SFRSTCR reg ON prgm.sgbstdn_pidm = reg.sfrstcr_pidm
JOIN
    SSBSECT sect ON reg.sfrstcr_crn = sect.ssbsect_crn
               AND reg.sfrstcr_term_code = sect.ssbsect_term_code
WHERE
    prgm.sgbstdn_term_code_eff > '202500'
    AND prgm.sgbstdn_stst_code = 'AS'
    AND prgm.sgbstdn_program_1 = 'AAS_SURG'
    AND prgm.sgbstdn_term_code_ctlg_1 < '202610'
    AND reg.sfrstcr_term_code = '202580'
    AND reg.sfrstcr_rsts_code LIKE 'R%'
    AND sect.ssbsect_subj_code = 'COMM'
    AND sect.ssbsect_crse_numb = '2025'
    AND pers.spriden_change_ind IS NULL
ORDER BY
    "Last Name", "First Name";
